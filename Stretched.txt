-- FTF | TELA ESTICADA
-- Menu local com controle de esticamento.
-- O valor é ajustável; o efeito real ainda precisa
-- de um método de renderização compatível.

local Players = game:GetService("Players")
local player = Players.LocalPlayer
local playerGui = player:WaitForChild("PlayerGui")

local old = playerGui:FindFirstChild("TelaEsticadaUI")
if old then old:Destroy() end

local gui = Instance.new("ScreenGui")
gui.Name = "TelaEsticadaUI"
gui.ResetOnSpawn = false
gui.Parent = playerGui

local frame = Instance.new("Frame")
frame.Size = UDim2.new(0, 260, 0, 225)
frame.Position = UDim2.new(0.5, -130, 0.5, -112)
frame.BackgroundColor3 = Color3.fromRGB(18, 18, 18)
frame.BorderSizePixel = 0
frame.Parent = gui

local corner = Instance.new("UICorner")
corner.CornerRadius = UDim.new(0, 10)
corner.Parent = frame

local function makeLabel(text, y, height)
    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(1, -20, 0, height)
    label.Position = UDim2.new(0, 10, 0, y)
    label.BackgroundTransparency = 1
    label.Text = text
    label.TextColor3 = Color3.new(1, 1, 1)
    label.TextScaled = true
    label.Font = Enum.Font.GothamBold
    label.Parent = frame
    return label
end

makeLabel("TELA ESTICADA | FTF", 8, 30)

local toggle = Instance.new("TextButton")
toggle.Size = UDim2.new(1, -20, 0, 36)
toggle.Position = UDim2.new(0, 10, 0, 45)
toggle.BackgroundColor3 = Color3.fromRGB(45, 45, 45)
toggle.TextColor3 = Color3.new(1, 1, 1)
toggle.Text = "Esticamento: DESLIGADO"
toggle.TextScaled = true
toggle.Font = Enum.Font.Gotham
toggle.Parent = frame

local enabled = false
toggle.Activated:Connect(function()
    enabled = not enabled
    toggle.Text = enabled
        and "Esticamento: LIGADO"
        or "Esticamento: DESLIGADO"
end)

makeLabel("Mudar esticamento: 100%", 90, 25)
local valueLabel = frame:FindFirstChildOfClass("TextLabel")

local slider = Instance.new("TextButton")
slider.Size = UDim2.new(1, -20, 0, 32)
slider.Position = UDim2.new(0, 10, 0, 120)
slider.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
slider.TextColor3 = Color3.new(1, 1, 1)
slider.Text = "Toque para mudar: 100%"
slider.TextScaled = true
slider.Parent = frame

local values = {75, 85, 100, 115, 125, 150}
local index = 3

slider.Activated:Connect(function()
    index = index % #values + 1
    slider.Text = "Toque para mudar: " .. values[index] .. "%"
    valueLabel.Text = "Mudar esticamento: " .. values[index] .. "%"
end)

local minimize = Instance.new("TextButton")
minimize.Size = UDim2.new(1, -20, 0, 30)
minimize.Position = UDim2.new(0, 10, 0, 170)
minimize.BackgroundTransparency = 1
minimize.TextColor3 = Color3.fromRGB(190, 190, 190)
minimize.Text = "Minimizar"
minimize.Parent = frame

local minimized = false
minimize.Activated:Connect(function()
    minimized = not minimized
    for _, item in ipairs(frame:GetChildren()) do
        if item ~= minimize and item:IsA("GuiObject") then
            item.Visible = not minimized
        end
    end
    frame.Size = minimized
        and UDim2.new(0, 130, 0, 38)
        or UDim2.new(0, 260, 0, 225)
    minimize.Text = minimized and "Abrir menu" or 
"Minimizar"
end)
